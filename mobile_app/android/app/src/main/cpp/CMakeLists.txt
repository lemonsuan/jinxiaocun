cmake_minimum_required(VERSION 3.22.1)
project(inventory_ocr)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if (NOT ANDROID_ABI STREQUAL "arm64-v8a")
    add_library(${PROJECT_NAME} SHARED unsupported_abi_stub.cpp)
    return()
endif()

set(ORT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/onnxruntime")
add_library(onnxruntime SHARED IMPORTED)
set_target_properties(onnxruntime PROPERTIES
    IMPORTED_LOCATION "${ORT_ROOT}/lib/arm64-v8a/libonnxruntime.so")
add_library(${PROJECT_NAME} SHARED ocr_v6_jni.cpp ocr_v6_engine.cpp)
target_include_directories(${PROJECT_NAME} PRIVATE "${ORT_ROOT}/include")
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra)
target_compile_options(${PROJECT_NAME} PRIVATE $<$<CONFIG:Release>:-O3> $<$<CONFIG:Release>:-fvisibility=hidden>)
target_link_libraries(${PROJECT_NAME} PRIVATE onnxruntime android log jnigraphics)
